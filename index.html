#decalre the dataset's url
dataset_path = "tmdb/tmdb-movie-metadata"
fite_name = "tmdb_5000_movies.csv" 
dataset_url = f"https://www.kaggle.com/api/v1/datasets/download/{dataset_path}/{fite_name}" 
data_sets_folder_name="data_sets"

#make HTTP get request to get the file using the url
get_file_response = requests.get(dataset_url) 
#raise an expetion if happens
get_file_response.raise_for_status() 

local_file_path = os.path.join(data_sets_folder_name, dataset_url.split('/')[-1]) 
zip_file_path = os.path.join(data_sets_folder_name, "dataset.zip")

with open(zip_file_path, mode = 'wb') as file:
    file.write(get_file_response.content) 

with zipfile.ZipFile(zip_file_path, 'r') as zip_ref:
    zip_ref.extractall(data_sets_folder_name)

# Clean up: delete the not needed zip file 
os.remove(zip_file_path)

This structure complicates analysis, as extracting meaningful information from these columns necessitates additional data processing steps. Parsing these stringified JSON columns into a more structured and analyzable format would significantly enhance the dataset's usability for analysis.
